from __future__ import annotations

from dataclasses import dataclass
from itertools import permutations, product
from math import factorial
from typing import Dict, List, Optional, Tuple

import os
import numpy as np
from scipy.optimize import linprog


Preference = List[int]
Profile = List[Preference]
ProfileKey = Tuple[Tuple[int, ...], ...]


def profile_to_key(profile: Profile) -> ProfileKey:
    return tuple(tuple(pref) for pref in profile)


def key_to_profile(key: ProfileKey) -> Profile:
    return [list(pref) for pref in key]


def all_preferences(n: int) -> List[Preference]:
    return [list(p) for p in permutations(range(n))]


def all_preference_profiles(n: int) -> List[Profile]:
    one_agent_prefs = all_preferences(n)
    return [[list(pref) for pref in prof] for prof in product(one_agent_prefs, repeat=n)]


def count_profiles(n: int) -> int:
    return factorial(n) ** n


def _rank_map(pref_i: Preference) -> Dict[int, int]:
    return {o: r + 1 for r, o in enumerate(pref_i)}


def _check_preferences(preferences: List[Preference]) -> None:
    n = len(preferences)
    for i, pref in enumerate(preferences):
        if len(pref) != n:
            raise ValueError(f"preferences[{i}] must have length {n}")
        if sorted(pref) != list(range(n)):
            raise ValueError(f"preferences[{i}] must be a permutation of 0..{n-1}")


def _check_doubly_stochastic_matrix(X: np.ndarray, name: str = "X", tol: float = 1e-8) -> None:
    if X.ndim != 2 or X.shape[0] != X.shape[1]:
        raise ValueError(f"{name} must be a square matrix, got shape={X.shape}")

    n = X.shape[0]

    if np.any(X < -tol):
        raise ValueError(f"{name} must satisfy {name} >= 0")

    row_sums = X.sum(axis=1)
    col_sums = X.sum(axis=0)

    if not np.allclose(row_sums, np.ones(n), atol=tol, rtol=0):
        raise ValueError(f"{name} row sums must all be 1. got {row_sums}")

    if not np.allclose(col_sums, np.ones(n), atol=tol, rtol=0):
        raise ValueError(f"{name} column sums must all be 1. got {col_sums}")


def _check_inputs(preferences: List[Preference], P: np.ndarray) -> None:
    _check_preferences(preferences)
    n = len(preferences)
    if P.shape != (n, n):
        raise ValueError(f"P must be {n}x{n}, got {P.shape}")
    _check_doubly_stochastic_matrix(P, name="P")


def vec(X: np.ndarray) -> np.ndarray:
    return np.asarray(X, dtype=float).reshape(-1, order="C")


def unvec(x: np.ndarray, n: int) -> np.ndarray:
    return np.asarray(x, dtype=float).reshape((n, n), order="C")


def uniform_assignment_matrix(n: int) -> np.ndarray:
    return np.full((n, n), 1.0 / n, dtype=float)


def s_matrix(preferences: List[Preference]) -> np.ndarray:
    n = len(preferences)
    S = np.zeros((n, n), dtype=float)
    for i in range(n):
        rm = _rank_map(preferences[i])
        for o in range(n):
            S[i, o] = n - rm[o] + 1
    return S


def s_vector(preferences: List[Preference]) -> np.ndarray:
    return vec(s_matrix(preferences))


def a_matrix(preferences: List[Preference]) -> np.ndarray:
    n = len(preferences)
    A = np.zeros((n * n, n * n), dtype=float)

    for i in range(n):
        pref_i = preferences[i]
        for r in range(1, n + 1):
            row = i * n + (r - 1)
            top_r = pref_i[:r]
            for o in top_r:
                col = i * n + o
                A[row, col] = 1.0

    return A


def b_matrix(n: int) -> np.ndarray:
    B = np.zeros((2 * n, n * n), dtype=float)

    for i in range(n):
        for o in range(n):
            B[i, i * n + o] = 1.0

    for o in range(n):
        for i in range(n):
            B[n + o, i * n + o] = 1.0

    return B


def cumulative_prob(row: np.ndarray, pref_i: Preference, r: int) -> float:
    top_r = pref_i[:r]
    return float(np.sum(row[top_r]))


def count_envy_violations(preferences: Profile, P: np.ndarray, tol: float = 1e-8) -> int:
    n = len(preferences)
    count = 0
    for i in range(n):
        pref_i = preferences[i]
        for j in range(n):
            for r in range(1, n + 1):
                own_val = cumulative_prob(P[i], pref_i, r)
                other_val = cumulative_prob(P[j], pref_i, r)
                if own_val + tol < other_val:
                    count += 1
    return count


def count_strategy_violations_at_profile(
    profile: Profile,
    p_map: Dict[ProfileKey, np.ndarray],
    one_agent_prefs: List[Preference],
    tol: float = 1e-8,
) -> int:
    n = len(profile)
    truthful_key = profile_to_key(profile)
    P_truth = p_map[truthful_key]
    count = 0

    for i in range(n):
        true_pref = profile[i]

        for misreport in one_agent_prefs:
            if misreport == true_pref:
                continue

            mis_profile = [list(pref) for pref in profile]
            mis_profile[i] = list(misreport)
            mis_key = profile_to_key(mis_profile)
            P_mis = p_map[mis_key]

            for r in range(1, n + 1):
                truthful_val = cumulative_prob(P_truth[i], true_pref, r)
                mis_val = cumulative_prob(P_mis[i], true_pref, r)
                if truthful_val + tol < mis_val:
                    count += 1

    return count


def check_sd_dominance(preferences: Profile, P: np.ndarray, Q: np.ndarray, tol: float = 1e-8) -> bool:
    n = len(preferences)
    for i in range(n):
        for r in range(1, n + 1):
            fq = cumulative_prob(Q[i], preferences[i], r)
            fp = cumulative_prob(P[i], preferences[i], r)
            if fq + tol < fp:
                return False
    return True


def pick_linprog_method(requested: Optional[str] = None) -> str:
    allowed = {"highs", "highs-ds", "highs-ipm", "interior-point", "revised simplex", "simplex"}
    if requested is None:
        return "highs"
    if requested not in allowed:
        raise ValueError(f"Unsupported linprog method={requested}. Choose one of {sorted(allowed)}")
    return requested


@dataclass
class LPResult:
    Q: np.ndarray
    q: np.ndarray
    objective: float
    status: str
    solver: str
    success: bool
    message: str
    nit: int
    A: np.ndarray
    B: np.ndarray
    s: np.ndarray


def solve_v_given_P(
    preferences: List[Preference],
    P: np.ndarray,
    solver: Optional[str] = None,
) -> LPResult:
    _check_inputs(preferences, P)

    n = len(preferences)
    A = a_matrix(preferences)
    B = b_matrix(n)
    s = s_vector(preferences)
    p = vec(P)
    b = np.ones(2 * n, dtype=float)

    c = -s.copy()
    constant_shift = float(s @ p)

    A_ub = -A
    b_ub = -(A @ p)

    A_eq = B
    b_eq = b

    bounds = [(0.0, None) for _ in range(n * n)]

    method = pick_linprog_method(solver)

    res = linprog(
        c=c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method=method,
    )

    if not res.success or res.x is None:
        raise RuntimeError(
            f"Inner LP failed. status={res.status}, method={method}, message={res.message}"
        )

    q_opt = np.asarray(res.x, dtype=float)
    Q_opt = unvec(q_opt, n)
    objective_value = float(s @ q_opt - constant_shift)

    return LPResult(
        Q=Q_opt,
        q=q_opt,
        objective=objective_value,
        status=str(res.status),
        solver=f"scipy.optimize.linprog(method='{method}')",
        success=bool(res.success),
        message=str(res.message),
        nit=int(getattr(res, "nit", -1)),
        A=A,
        B=B,
        s=s,
    )


def _set_single_thread_env() -> None:
    """
    子プロセス内で BLAS / OpenMP の過剰並列を抑える。
    """
    os.environ["OMP_NUM_THREADS"] = "1"
    os.environ["OPENBLAS_NUM_THREADS"] = "1"
    os.environ["MKL_NUM_THREADS"] = "1"
    os.environ["VECLIB_MAXIMUM_THREADS"] = "1"
    os.environ["NUMEXPR_NUM_THREADS"] = "1"


def solve_inner_for_profile_key(
    key: ProfileKey,
    P: np.ndarray,
    solver: Optional[str] = None,
) -> Tuple[ProfileKey, np.ndarray, float, str, str, int]:
    _set_single_thread_env()
    profile = key_to_profile(key)
    res = solve_v_given_P(profile, P, solver=solver)
    return key, res.q, float(res.objective), res.status, res.message, int(res.nit)


def solve_inner_chunk(
    chunk: List[Tuple[ProfileKey, np.ndarray]],
    solver: Optional[str] = None,
) -> List[Tuple[ProfileKey, np.ndarray, float, str, str, int]]:
    """
    chunk 単位で inner LP を連続実行する worker。
    """
    _set_single_thread_env()
    out: List[Tuple[ProfileKey, np.ndarray, float, str, str, int]] = []
    for key, P in chunk:
        profile = key_to_profile(key)
        res = solve_v_given_P(profile, P, solver=solver)
        out.append((key, res.q, float(res.objective), res.status, res.message, int(res.nit)))
    return out


@dataclass
class MasterResult:
    p_map: Dict[ProfileKey, np.ndarray]
    t_map: Dict[ProfileKey, float]
    objective: float
    success: bool
    status: str
    message: str
    nit: int
    solver: str


def solve_master_problem(
    profiles: List[Profile],
    cut_sets: Dict[ProfileKey, List[np.ndarray]],
    solver: Optional[str] = None,
    tol: float = 1e-9,
) -> MasterResult:
    if not profiles:
        raise ValueError("profiles must be non-empty")

    n = len(profiles[0])
    m = len(profiles)
    one_agent_prefs = all_preferences(n)
    profile_keys = [profile_to_key(p) for p in profiles]
    key_to_index = {k: idx for idx, k in enumerate(profile_keys)}

    p_block_size = n * n
    num_p_vars = m * p_block_size
    num_t_vars = m
    total_vars = num_p_vars + num_t_vars

    def p_offset(theta_idx: int) -> int:
        return theta_idx * p_block_size

    def t_offset(theta_idx: int) -> int:
        return num_p_vars + theta_idx

    c = np.zeros(total_vars, dtype=float)
    for theta_idx in range(m):
        c[t_offset(theta_idx)] = 1.0

    A_eq_rows = []
    b_eq_rows = []

    for theta_idx in range(m):
        base = p_offset(theta_idx)

        for i in range(n):
            row = np.zeros(total_vars, dtype=float)
            for o in range(n):
                row[base + i * n + o] = 1.0
            A_eq_rows.append(row)
            b_eq_rows.append(1.0)

        for o in range(n):
            row = np.zeros(total_vars, dtype=float)
            for i in range(n):
                row[base + i * n + o] = 1.0
            A_eq_rows.append(row)
            b_eq_rows.append(1.0)

    A_ub_rows = []
    b_ub_rows = []

    for theta_idx, profile in enumerate(profiles):
        base = p_offset(theta_idx)
        for i in range(n):
            pref_i = profile[i]
            for j in range(n):
                for r in range(1, n + 1):
                    row = np.zeros(total_vars, dtype=float)
                    top_r = pref_i[:r]
                    for o in top_r:
                        row[base + i * n + o] -= 1.0
                        row[base + j * n + o] += 1.0
                    A_ub_rows.append(row)
                    b_ub_rows.append(0.0)

    for theta_idx, profile in enumerate(profiles):
        truth_base = p_offset(theta_idx)

        for i in range(n):
            true_pref = profile[i]

            for misreport in one_agent_prefs:
                if misreport == true_pref:
                    continue

                mis_profile = [list(pref) for pref in profile]
                mis_profile[i] = list(misreport)
                mis_key = profile_to_key(mis_profile)
                mis_idx = key_to_index[mis_key]
                mis_base = p_offset(mis_idx)

                for r in range(1, n + 1):
                    row = np.zeros(total_vars, dtype=float)
                    top_r = true_pref[:r]
                    for o in top_r:
                        row[truth_base + i * n + o] -= 1.0
                        row[mis_base + i * n + o] += 1.0
                    A_ub_rows.append(row)
                    b_ub_rows.append(0.0)

    for theta_idx, profile in enumerate(profiles):
        key = profile_keys[theta_idx]
        base = p_offset(theta_idx)
        t_idx = t_offset(theta_idx)
        s_theta = s_vector(profile)

        for q_cut in cut_sets[key]:
            row = np.zeros(total_vars, dtype=float)
            row[base: base + p_block_size] = -s_theta
            row[t_idx] = -1.0
            rhs = -float(s_theta @ q_cut)
            A_ub_rows.append(row)
            b_ub_rows.append(rhs)

    A_eq = np.vstack(A_eq_rows) if A_eq_rows else None
    b_eq = np.array(b_eq_rows, dtype=float) if b_eq_rows else None
    A_ub = np.vstack(A_ub_rows) if A_ub_rows else None
    b_ub = np.array(b_ub_rows, dtype=float) if b_ub_rows else None

    bounds = [(0.0, None) for _ in range(num_p_vars)] + [(0.0, None) for _ in range(num_t_vars)]

    method = pick_linprog_method(solver)
    res = linprog(
        c=c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method=method,
    )

    if not res.success or res.x is None:
        raise RuntimeError(
            f"Master LP failed. status={res.status}, method={method}, message={res.message}"
        )

    x = np.asarray(res.x, dtype=float)
    p_map: Dict[ProfileKey, np.ndarray] = {}
    t_map: Dict[ProfileKey, float] = {}

    for theta_idx, key in enumerate(profile_keys):
        base = p_offset(theta_idx)
        p_map[key] = unvec(x[base: base + p_block_size], n)
        t_map[key] = float(x[t_offset(theta_idx)])

    return MasterResult(
        p_map=p_map,
        t_map=t_map,
        objective=float(res.fun),
        success=bool(res.success),
        status=str(res.status),
        message=str(res.message),
        nit=int(getattr(res, "nit", -1)),
        solver=f"scipy.optimize.linprog(method='{method}')",
    )

def matrix_to_jsonable(X: np.ndarray, digits: int = 12) -> List[List[float]]:
    return [[round(float(v), digits) for v in row] for row in X.tolist()]


def matrix_to_string(X: np.ndarray, digits: int = 12) -> str:
    return json.dumps(matrix_to_jsonable(X, digits=digits), ensure_ascii=False)


def profile_to_string(profile: Profile) -> str:

def print_matrix(name: str, X: np.ndarray, digits: int = 6) -> None:
    print(f"{name}:")
    for row in X:
        print("  ".join(f"{v:.{digits}f}" for v in row))
    print()


def print_profile(profile: Profile) -> None:

def write_csv_header(writer: csv.DictWriter) -> None:
    writer.writeheader()


def build_chunks(
    p_map: Dict[ProfileKey, np.ndarray],
    chunk_size: int,
) -> List[List[Tuple[ProfileKey, np.ndarray]]]:
    items = list(p_map.items())
    return [items[i:i + chunk_size] for i in range(0, len(items), chunk_size)]


def solve_inner_batch_parallel_chunked(
    p_map: Dict[ProfileKey, np.ndarray],
    max_workers: int | None = None,
    chunk_size: int = 256,
    mp_start_method: str = "spawn",
) -> Dict[ProfileKey, Tuple[np.ndarray, float, str, str, int]]:
    """
    inner LP を chunk 単位で並列実行する。
    戻り値:
      key -> (q_opt, objective, status, message, nit)
    """
    chunks = build_chunks(p_map, chunk_size=chunk_size)
    results: Dict[ProfileKey, Tuple[np.ndarray, float, str, str, int]] = {}

    ctx = get_context(mp_start_method)

    with ProcessPoolExecutor(max_workers=max_workers, mp_context=ctx) as ex:
        futures = [ex.submit(solve_inner_chunk, chunk, None) for chunk in chunks]

        completed = 0
        total = len(futures)

        for fut in as_completed(futures):
            batch = fut.result()
            for key, q_opt, obj, status, message, nit in batch:
                results[key] = (q_opt, obj, status, message, nit)

            completed += 1
            if completed % 10 == 0 or completed == total:
                print(f"  inner chunk solved: {completed}/{total}")

    return results

    master_result = None
    last_inner_results: Dict[ProfileKey, Tuple[np.ndarray, float, str, str, int]] = {}

    for iteration in range(1, max_iterations + 1):

                "profile_id": profile_id,
                "preferences": profile_to_string(profile),
                "vP": f"{float(v_theta):.12f}",
                "t_theta": f"{float(t_theta):.12f}",
                "sd_ok": sd_ok,
                "direct_value": f"{direct_value:.12f}",

                "envy_violation_count": envy_violation_count,
                "envy_violation_is_zero": envy_violation_is_zero,
                "strategy_violation_count": strategy_violation_count,
                "strategy_violation_is_zero": strategy_violation_is_zero,
                "P": matrix_to_string(P_theta),
                "Q": matrix_to_string(Q_theta),
            }

            writer.writerow(record)

            if best_record is None or float(v_theta) > float(best_record["vP"]):
                best_record = {
                    **record,
                    "P_matrix": P_theta,
                    "Q_matrix": Q_theta,
                    "profile_obj": profile,
                }

            if profile_id % 5000 == 0 or profile_id == len(profiles):
                print(f"wrote {profile_id}/{len(profiles)} profiles to CSV")

    print()

    print(f"sd-efficient profiles   (v(P) <= {tol}): {num_sd_efficient}")
